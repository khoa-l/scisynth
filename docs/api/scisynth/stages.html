30    ------
31    ValueError
32        If ``sigma`` is negative.
33
34    Notes
35    -----
36    A full-shape array is always drawn, so the noise at a given position does not

49        return obs.replace(values=np.where(obs.valid, obs.values + noise, obs.values))

82    def apply(
83        self, obs: Observation, rng: np.random.Generator, params: Mapping[str, Any]
84    ) -> Observation:
85        scale = float(params["scale"])
86        if scale <= 0:
87            raise ValueError(f"scale must be > 0, got {scale}")

18@dataclass
19class Quantize(Stage):
20    """Round valid values to evenly spaced levels, like an ADC.
21
22    Parameters
23    ----------
24    levels : int, default=256
25        Number of levels; at least 2.
26    vmin, vmax : float or array-like, optional
27        Range covered by the levels. By default it is the range of the valid data,
28        taken per channel. Values outside are clipped to the end levels. Give one
29        value per channel to set per-channel ranges.
30
31    Raises
32    ------
33    ValueError
34        If ``levels`` is less than 2, or ``vmin`` is not below ``vmax``.
35
36    Notes
37    -----
38    The result stays float: the level values, not integer codes. A channel whose
39    data range is empty is left unchanged. It uses no randomness.
40    """
41
42    levels: Param = 256
43    vmin: Param | None = None
44    vmax: Param | None = None
45
46    def demo_observation(self) -> Observation:
47        """Return a monotone gradient, so levels show up as discrete bands.
48
49        Returns
50        -------
51        Observation
52            A 2-D grid with a diagonal gradient.
53        """
54        from ..demo import ramp
55
56        return ramp()
57
58    def apply(
59        self, obs: Observation, rng: np.random.Generator, params: Mapping[str, Any]
60    ) -> Observation:
61        levels = as_int(params["levels"], "levels")
62        if levels < 2:
63            raise ValueError(f"levels must be >= 2, got {levels}")
64        if (
65            params["vmin"] is not None
66            and params["vmax"] is not None
67            and np.any(np.asarray(params["vmin"]) >= np.asarray(params["vmax"]))
68        ):
69            raise ValueError(
70                f"vmin must be below vmax, got vmin={params['vmin']!r}, "
71                f"vmax={params['vmax']!r}"
72            )
73        valid = obs.values[obs.mask]  # (n_valid, d)
74        if valid.shape[0] == 0:
75            return obs.replace()
76        lo = (
77            valid.min(axis=0)
78            if params["vmin"] is None
79            else np.asarray(params["vmin"], float)
80        )
81        hi = (
82            valid.max(axis=0)
83            if params["vmax"] is None
84            else np.asarray(params["vmax"], float)
85        )
86        flat = hi <= lo
87        step = np.where(flat, 1.0, (hi - lo) / (levels - 1))
88        codes = np.clip(np.round((obs.values - lo) / step), 0, levels - 1)
89        quantized = np.where(flat, obs.values, lo + codes * step)
90        return obs.replace(values=np.where(obs.valid, quantized, obs.values))

 99    """Reduce resolution by averaging nearby locations.
100
101    On a grid, each output is the mean of the valid inputs in a block of cells. On
102    points, the bounding box is cut into bins and each occupied bin gives one output
103    point, so the result is a flat list of points. In both, the output's coordinates
104    are the mean of its inputs' coordinates and ``truth`` is averaged over all of
105    them.
106
107    Parameters
108    ----------
109    factor : int, sequence of int or distribution, default=2
110        Block size along every axis, or one per axis. For points it divides the
111        resolution of a grid with as many points as there are locations.
112
113    Raises
114    ------
115    ValueError
116        If ``factor`` has the wrong length or is not a positive integer, or a valid
117        location has a non-finite coordinate.
118
119    Notes
120    -----
121    An output location is valid if any input in its bin is. At the edge of a grid
122    whose size is not a multiple of ``factor``, the last block is smaller. The
123    outputs get new ids above the current maximum; :meth:`links_from` ties each of them
124    to the inputs it averaged, with weight ``1 / n_valid``.
125    """

130        self, obs: Observation, rng: np.random.Generator, params: Mapping[str, Any]
131    ) -> Observation:
132        factors = _factors(params["factor"], len(obs.coords))
133        placed(obs)
134        obs, bins, shape, on_grid = _binning(obs, factors)
135        n, d, n_bins = obs.size, obs.n_channels, math.prod(shape)
136        weights = _average(bins, n_bins, obs.mask)
137        everywhere = _average(bins, n_bins, np.ones(n, dtype=np.bool_))
138        mask = np.asarray(weights.getnnz(axis=1) > 0)
139        values = weights @ obs.values.reshape(n, d)
140        values[~mask] = np.nan
141        truth = None
142        if obs.truth is not None:

144        if on_grid:
145            coords = {}
146            for (name, coord), f in zip(obs.coords.items(), factors, strict=True):
147                block = np.arange(len(coord)) // f
148                coords[name] = np.bincount(block, weights=coord) / np.bincount(block)
149        else:
