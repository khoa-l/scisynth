scisynth.observer

Observers: a sampler and stages run as one instrument, and the record of a run.

Holds the Observer, its StageList, and the Trace of one run with the StageDiff of each stage. These names are importable from the package itself, for example from scisynth.observer import Observer, Trace.

142            The output after the sampler and after each stage, so

145        """
146        rngs = spawn_rngs(seed, 1 + len(self.stages))
147        obs = _named(self.sampler(latent, rngs[0]), self.sampler_name)
148        out, links = [obs], []
149        for (name, stage), rng in zip(self.stages.items(), rngs[1:], strict=True):
150            after, params = stage.call_with_params(obs, rng)
151            after = _named(after, name)
152            links.append(stage.links_from(obs, after, params))
153            out.append(obs := after)
154        return Trace(out, links)

class StageList(scisynth.observer.named_sequence.NamedSequence[scisynth.stages.base.Stage]): 
 57class StageList(NamedSequence[Stage]):
 58    """An ordered sequence of stages, each with a unique name.
 59
 60    This is what :attr:`Observer.stages <scisynth.observer.observer.Observer.stages>`
 61    holds. Iterating gives the stages; use :meth:`items` for ``(name, stage)``
 62    pairs.
 63
 64    Parameters
 65    ----------
 66    stages : iterable of Stage or (str, Stage), default=()
 67        A bare stage is named after its class in snake_case (``GaussianNoise`` ->
 68        ``"gaussian_noise"``); if a class appears more than once, or the name is
 69        taken, they are numbered ``"gaussian_noise_1"``, ``"gaussian_noise_2"``, ...
 70        A ``StageList`` keeps its names.
 71    reserved : iterable of str, default=()
 72        Names the stages may not use, e.g. the sampler's.
 73
 74    Attributes
 75    ----------
 76    names : list of str
 77        The stage names, in order.
 78
 79    Raises
 80    ------
 81    TypeError
 82        If an element is not a Stage.
 83    ValueError
 84        If a name is empty, contains ``"__"`` (reserved) or is repeated.
 85
 86    Notes
 87    -----
 88    ``stages[i]`` is the ``i``-th stage (negative counts from the end) and
 89    ``stages["name"]`` the stage with that name. A slice returns a ``StageList``
 90    with the names kept; slice bounds are positions, not names.
 91
 92    Examples
 93    --------
 94    >>> from scisynth.observer import StageList
 95    >>> from scisynth.stages import GaussianNoise, RandomDropout
 96    >>> stages = StageList([GaussianNoise(0.1), ("drop", RandomDropout(0.1))])
 97    >>> stages.names
 98    ['gaussian_noise', 'drop']
 99    >>> stages[1:].names
100    ['drop']
101    """
102
103    _kind = "stage"
104
105    def __init__(
106        self,
107        stages: Iterable[StageSpec] = (),
108        *,
109        reserved: Iterable[str] = (),
110    ) -> None:
111        specs = stages.items() if isinstance(stages, StageList) else stages
112        explicit: list[str | None] = []
113        parsed: list[Stage] = []
114        for spec in specs:
115            name: str | None = None
116            if isinstance(spec, tuple):
117                name, spec = spec
118                check_name(name)
119            if not isinstance(spec, Stage):
120                raise TypeError(
121                    f"stages must be Stage instances, got {type(spec).__name__}"
122                )
123            explicit.append(name)
124            parsed.append(spec)
125
126        taken = [*reserved, *(n for n in explicit if n is not None)]
127        duplicates = sorted(n for n, c in Counter(taken).items() if c > 1)
128        if duplicates:
129            raise ValueError(f"Names must be unique; repeated: {duplicates}")
130        self._stages = tuple(parsed)
131        self._set_names(_autoname(parsed, explicit, set(taken)))
