 84class InterpolatedField(FieldOps):
 85    """A field stored on a regular grid and interpolated between nodes.
 86
 87    Parameters
 88    ----------
 89    values : ndarray of shape (n_0, ..., n_{ndim-1})
 90        Field values at the nodes. Node ``i`` on an axis with extent ``(lo, hi)``
 91        and ``n`` nodes sits at ``lo + i * (hi - lo) / n``, so the last node is one
 92        spacing short of ``hi``.
 93    domain : Domain
 94        The space the field lives in.
 95    order : int, default=1
 96        Spline order, from 0 (nearest) and 1 (linear) up to 5.
 97    mode : str, default="nearest"
 98        A :func:`scipy.ndimage.map_coordinates` boundary mode. With ``"nearest"``,
 99        queries beyond the outermost nodes (including the last cell next to ``hi``)
100        take the edge value; ``"grid-wrap"`` makes the field periodic.
101
102    Raises
103    ------
104    ValueError
105        If ``order`` is outside 0-5 or ``values`` does not have one dimension per
106        axis.
107
108    Notes
109    -----
110    The field has one channel.
111    """
112
113    values: FloatArray
114    domain: Domain
115    order: int = 1
116    mode: str = "nearest"
117    kind: ClassVar[Kind] = "field"
118
119    def __post_init__(self) -> None:
120        if not 0 <= self.order <= 5:
121            raise ValueError(f"order must be between 0 and 5, got {self.order}")
122        self.values = np.asarray(self.values, dtype=np.float64)
123        if self.values.ndim != self.domain.ndim:
124            raise ValueError(
125                f"values has {self.values.ndim} dims, domain has {self.domain.ndim}"
126            )
127
128    def evaluate(self, points: FloatArray) -> FloatArray:
129        """Interpolate at ``points``.
130
131        Parameters
132        ----------
133        points : ndarray of shape (n, ndim)
134            Query points, one row per point.
135
136        Returns
137        -------
138        ndarray of shape (n, 1)
139            The interpolated values.
140        """
141        idx = np.stack(
142            [
143                (points[:, i] - lo) / ((hi - lo) / n)
144                for i, ((lo, hi), n) in enumerate(
145                    zip(self.domain.extents, self.values.shape, strict=True)
146                )
147            ]
148        )
149        out: FloatArray = ndimage.map_coordinates(
150            self.values, idx, order=self.order, mode=self.mode
151        )
152        return out[:, None]

156class GaussianField(LatentGenerator):
157    """Stationary, periodic Gaussian random field with an RBF covariance.
158
159    Parameters
160    ----------
161    domain : Domain
162        The space the field lives in.
163    length_scale : float or distribution, default=0.1
164        The RBF length scale ``l``; correlation falls to 1/e at ``r = sqrt(2) * l``.
165        Must be positive. A distribution gives one draw per realization.
166    amplitude : float or distribution, default=1.0
167        Standard deviation of the field.
168    mean : float or distribution, default=0.0
169        Mean of the field.
170    resolution : int or sequence of int, default=128
171        Grid nodes per axis (an int is repeated for every axis). The field holds no
172        detail finer than this grid.
173    order : int, default=1
174        Interpolation spline order when querying: 1 (linear) shows kinks at the
175        nodes, 3 (cubic) is smooth, 0 is nearest-neighbor.
176    max_cells : int, default=50_000_000
177        Upper limit on ``prod(resolution)``, checked before allocating.
178
179    Notes
180    -----
181    A realization is white noise filtered in Fourier space by ``exp(-k^2 l^2 / 4)``,
182    which gives the covariance ``exp(-r^2 / (2 l^2))``, the same kernel as
183    scikit-learn's ``RBF(length_scale=l)``. The result is standardized to zero mean
184    and unit variance, then scaled to ``mean + amplitude * field``. The FFT makes it
185    periodic, so the realization wraps around the domain.
186
187    The grid has ``resolution ** ndim`` cells, so higher-dimensional domains need a
188    small ``resolution``.
189
190    Examples
191    --------
192    >>> import numpy as np
193    >>> from scisynth.core import Domain
194    >>> from scisynth.latent import GaussianField
195    >>> domain = Domain.from_extents([(0, 10), (0, 10)])
196    >>> latent = GaussianField(domain, length_scale=1.5).realize(seed=1)
197    >>> latent.evaluate(np.array([[5.0, 5.0]])).shape
198    (1, 1)
199    """
200
201    domain: Domain
202    length_scale: Param = 0.1
203    amplitude: Param = 1.0
204    mean: Param = 0.0
205    resolution: int | Sequence[int] = 128
206    order: int = 1
207    max_cells: int = MAX_FIELD_CELLS

209    def realize(self, seed: SeedLike = None) -> InterpolatedField:
210        """Draw one realization.
211
212        Parameters
213        ----------
214        seed : int, SeedSequence or None, default=None
215            Seed for the random streams; None draws fresh OS entropy.
216
217        Returns
218        -------
219        InterpolatedField
220            One channel, periodic (``mode="grid-wrap"``).
221
222        Raises
223        ------
224        ValueError
225            If ``length_scale`` is not positive or the grid exceeds ``max_cells``.
226        """
227        param_rng, noise_rng = spawn_rngs(seed, 2)
228        params = self.resolved_params(param_rng)
229        shape = as_shape(params["resolution"], self.domain.ndim, "resolution")
230        check_cells(shape, self.max_cells, "resolution")
231        length = float(params["length_scale"])
232        if length <= 0:
233            raise ValueError(f"length_scale must be > 0, got {length}")
234
235        freqs = [
236            2 * np.pi * np.fft.fftfreq(n, d=(hi - lo) / n)
237            for n, (lo, hi) in zip(shape, self.domain.extents, strict=True)
238        ]
239        k2 = np.sum([g**2 for g in np.meshgrid(*freqs, indexing="ij")], axis=0)
240        white = noise_rng.standard_normal(shape)
241        field = np.fft.ifftn(np.fft.fftn(white) * np.exp(-0.25 * length**2 * k2)).real
242        field -= field.mean()
243        std = field.std()
244        if std > 0:
245            field /= std
246        return InterpolatedField(
247            float(params["mean"]) + float(params["amplitude"]) * field,
248            self.domain,
249            order=as_int(params["order"], "order"),
250            mode="grid-wrap",  # the FFT construction is periodic
251        )

232        if length <= 0:
233            raise ValueError(f"length_scale must be > 0, got {length}")
234
235        freqs = [
236            2 * np.pi * np.fft.fftfreq(n, d=(hi - lo) / n)
237            for n, (lo, hi) in zip(shape, self.domain.extents, strict=True)
238        ]
239        k2 = np.sum([g**2 for g in np.meshgrid(*freqs, indexing="ij")], axis=0)
240        white = noise_rng.standard_normal(shape)
241        field = np.fft.ifftn(np.fft.fftn(white) * np.exp(-0.25 * length**2 * k2)).real
242        field -= field.mean()
243        std = field.std()
244        if std > 0:
245            field /= std
246        return InterpolatedField(
247            float(params["mean"]) + float(params["amplitude"]) * field,
248            self.domain,
249            order=as_int(params["order"], "order"),
250            mode="grid-wrap",  # the FFT construction is periodic
251        )

139        corr: Sequence[Sequence[float]] | FloatArray | None = None,
140    ) -> None:

145        self.corr: FloatArray | None = None
146        self._mix: FloatArray | None = None
147
148        if corr is not None:

150            d = len(channels)

152                raise ValueError(f"corr must be {d}x{d}, got shape {matrix.shape}")
153            if not (np.allclose(matrix, matrix.T) and np.allclose(np.diag(matrix), 1)):
154                raise ValueError("corr must be symmetric with a unit diagonal")
155            try:

157            except np.linalg.LinAlgError:

159            self.corr = matrix
