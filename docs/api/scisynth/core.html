scisynth.core

Core building blocks: observations, domains, components, RNG, parameters.

This package depends on nothing else in scisynth, and everything else depends on it, so it must stay that way: importing it never needs another subpackage.

Its public names are importable from the package itself, for example from scisynth.core import Domain, Observation.

23def resolve(param: Any, rng: np.random.Generator) -> Any:
24    """Turn a parameter specification into a concrete value.
25
26    Parameters
27    ----------
28    param : object
29        An object with an ``rvs`` method (such as a frozen ``scipy.stats``
30        distribution) is drawn once with ``param.rvs(random_state=rng)``. Any other
31        callable is called as ``param(rng)``. Everything else (numbers, arrays,
32        strings, tuples, None) is returned unchanged.
33    rng : numpy.random.Generator
34        Source of randomness for distributions and callables.
35
36    Returns
37    -------
38    object
39        The concrete value.
40
41    Examples
42    --------
43    >>> import numpy as np
44    >>> from scipy import stats
45    >>> from scisynth.core import resolve
46    >>> rng = np.random.default_rng(0)
47    >>> resolve(0.5, rng)
48    0.5
49    >>> round(float(resolve(stats.uniform(0, 1), rng)), 3)  # one draw
50    0.637
51    """
52    if isinstance(param, HasRvs):
53        return param.rvs(random_state=rng)
54    if callable(param):
55        return param(rng)
56    return param

def spawn_rngs( seed: int | Sequence[int] | numpy.random.bit_generator.SeedSequence | None, n: int) -> list[numpy.random._generator.Generator]: 
27def spawn_rngs(seed: SeedLike, n: int) -> list[np.random.Generator]:
28    """Spawn independent random generators from one seed.
29
30    Parameters
31    ----------
32    seed : int, sequence of int, SeedSequence or None
33        None draws fresh OS entropy. A ``Generator`` is rejected, since sharing one
34        would couple otherwise independent streams. A ``SeedSequence`` is never
35        consumed: repeated calls give identical children.
36    n : int
37        Number of generators; must be non-negative.
38
39    Returns
40    -------
41    list of numpy.random.Generator
42        Child ``i`` depends only on ``(seed, i)``, never on ``n``, so asking for
43        more generators does not change the earlier ones.
44
45    Raises
46    ------
47    TypeError
48        If ``seed`` is a ``Generator``.
49    ValueError
50        If ``n`` is negative.
51    """
52    if n < 0:
53        raise ValueError(f"n must be non-negative, got {n}")
54    return [np.random.default_rng(child) for child in _root(seed).spawn(n)]

  • n (int): Number of generators; must be non-negative.
